// ============================================= // CRM Manager - Google Apps Script バックエンド // CORS対応・GET経由での保存・削除対応版 // ============================================= const SHEET_NAMES = { customers: '顧客', projects: '案件', payments: '入金', history: '対応履歴' }; const HEADERS = { customers: ['id','name','company','tel','email','address','invoiceNum','note','createdAt'], projects: ['id','name','customerId','customerName','desc','amount','status'], payments: ['id','customerId','customerName','billed','dueDate','paidDate','paid','unpaid'], history: ['id','date','customerId','customerName','staff','content','nextDate'] }; // ============================================= // GETリクエスト処理 // POSTがCORSで失敗するため、全操作をGETで処理 // ============================================= function doGet(e) { try { const action = e.parameter.action; const sheet = e.parameter.sheet; // データ全件取得 if (action === 'getAll') { const data = getAllRows(sheet); return buildResponse({ status: 'ok', data: data }); } // データ保存（dataパラメータにJSON文字列を渡す） if (action === 'save') { const record = JSON.parse(decodeURIComponent(e.parameter.data)); const result = saveRow(sheet, record); return buildResponse({ status: 'ok', data: result }); } // データ削除 if (action === 'delete') { deleteRow(sheet, e.parameter.id); return buildResponse({ status: 'ok' }); } return buildResponse({ status: 'error', message: '不明なアクション' }); } catch(err) { return buildResponse({ status: 'error', message: err.message }); } } // ============================================= // シート取得（なければ作成＋ヘッダー追加） // ============================================= function getOrCreateSheet(sheetName) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); const key = Object.keys(SHEET_NAMES).find(k => SHEET_NAMES[k] === sheetName); if (key && HEADERS[key]) { sheet.appendRow(HEADERS[key]); const range = sheet.getRange(1, 1, 1, HEADERS[key].length); range.setFontWeight('bold'); range.setBackground('#1a56db'); range.setFontColor('#ffffff'); } } return sheet; } // ============================================= // 全行取得 // ============================================= function getAllRows(sheetKey) { const sheetName = SHEET_NAMES[sheetKey]; if (!sheetName) throw new Error('不明なシート: ' + sheetKey); const sheet = getOrCreateSheet(sheetName); const values = sheet.getDataRange().getValues(); if (values.length <= 1) return []; const headers = values[0]; return values.slice(1).map(row => { const obj = {}; headers.forEach((h, i) => { obj[h] = row[i]; }); return obj; }); } // ============================================= // 行を保存（新規 or 更新） // ============================================= function saveRow(sheetKey, record) { const sheetName = SHEET_NAMES[sheetKey]; if (!sheetName) throw new Error('不明なシート: ' + sheetKey); const sheet = getOrCreateSheet(sheetName); const headers = HEADERS[sheetKey]; const values = sheet.getDataRange().getValues(); const idCol = headers.indexOf('id'); let foundRow = -1; for (let i = 1; i < values.length; i++) { if (String(values[i][idCol]) === String(record.id)) { foundRow = i + 1; break; } } const rowData = headers.map(h => record[h] !== undefined ? record[h] : ''); if (foundRow > 0) { sheet.getRange(foundRow, 1, 1, rowData.length).setValues([rowData]); } else { sheet.appendRow(rowData); } return record; } // ============================================= // 行を削除 // ============================================= function deleteRow(sheetKey, id) { const sheetName = SHEET_NAMES[sheetKey]; if (!sheetName) throw new Error('不明なシート: ' + sheetKey); const sheet = getOrCreateSheet(sheetName); const headers = HEADERS[sheetKey]; const values = sheet.getDataRange().getValues(); const idCol = headers.indexOf('id'); for (let i = values.length - 1; i >= 1; i--) { if (String(values[i][idCol]) === String(id)) { sheet.deleteRow(i + 1); break; } } } // ============================================= // JSONレスポンス // ============================================= function buildResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
